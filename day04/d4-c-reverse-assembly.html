00 总览：四种结构在 32 位下的本质 先记结论，再往下看证据
全局变量
躺在 .data 段固定地址，汇编 mov [0x402000] 直接访存，程序全程有效
局部变量
活在栈上 [ebp-4]，随函数进出而分配/销毁，汇编用 ebp 负偏移访问
数组
栈上连续内存，下标即 基址 + i×4，汇编出现 [ebp+eax*4-0x18] 比例变址
指针
存"地址"的变量。传参传地址（lea），解引用是"取地址再访存"两跳

.data 段（程序镜像内，地址固定）
0x402000g_counter = 100
0x402004g_hidden = 7
运行时栈（esp → ebp 向下增长，函数帧内）
[ebp+0xC]arg2(调用者压入)← 调用者写
[ebp+0x8]arg1(调用者压入)← 参数
[ebp+0x4]返回地址(call 压入)
[ebp+0x0]旧 ebp(push ebp)
[ebp-0x4]局部变量int local = 42
[ebp-0x8]变量 i / s(数组示例)
[ebp-0x18]arr[0] … arr[3]连续 16 字节
[ebp-0x4]指针变量 p里面存"别人"的地址
一句话规律：ebp 往上是"别人的地盘"（参数、返回地址），ebp 往下是本函数的局部区。全局变量根本不在这条栈上，它有自己独立的固定地址。
01 D4-1 局部变量 vs 全局变量 d1_vars.c → d1_vars.exe @ 0x401000

程序（节选）

/* 已初始化全局 → .data 段，地址固定 */
int g_counter = 100;

int add_global(int delta) {
    g_counter += delta;   // 直接读写 [0x402000]
    return g_counter;
}

int add_local(int delta) {
    int local = 42;      // 局部变量 → 栈 [ebp-4]
    local += delta;
    return local;
}
32 位反汇编（Intel 语法）
00401000 <_add_global>:
  push  ebp
  mov   ebp, esp
  mov   edx, DWORD PTR ds:0x402000   ; 读全局：固定地址直访
  mov   eax, DWORD PTR [ebp+0x8]    ; 参数 delta 从栈取
  add   eax, edx
  mov   ds:0x402000, eax            ; 写回全局
  mov   eax, ds:0x402000
  pop   ebp
  ret

0040101a <_add_local>:
  push  ebp
  mov   ebp, esp
  sub   esp, 0x4                     ; 给局部变量腾 4 字节
  mov   DWORD PTR [ebp-0x4], 0x2a   ; local = 42 (0x2a)
  mov   eax, DWORD PTR [ebp+0x8]
  add   DWORD PTR [ebp-0x4], eax      ; local += delta
  mov   eax, DWORD PTR [ebp-0x4]
  leave
  ret

参数为何在 [ebp+0x8]？call 前调用者把 delta 压栈（这里是 mov [esp],0x5），call 压入返回地址，push ebp 后再设 ebp。于是 ebp+8 恰好是第一个参数，ebp+4 是返回地址。这是 x86 32 位函数调用的通用契约。
逆向识别点：看到 mov reg, ds:[固定地址] 且地址在镜像内（如 0x400000~0x500000 之间）→ 这是全局/静态变量；看到 [ebp-0xN] 负偏移 → 局部变量；[ebp+0xN] 正偏移 → 参数。目标文件里全局地址写作 ds:0x0 + dir32 .data 重定位，链接时才填成 0x402000。
02 D4-2 数组：栈上连续内存 + 下标寻址 d2_array.c → d2_array.exe @ 0x401000

程序（节选）

int sum_array(int n) {
    int arr[4] = {10, 20, 30, 40};  // 栈上 4×4 = 16 字节
    int s = 0, i;
    for (i = 0; i < 4; i++)
        s += arr[i];              // 下标 → 基址 + i×4
    return s + n;
}
32 位反汇编（节选）
00401000 <_sum_array>:
  push  ebp
  mov   ebp, esp
  sub   esp, 0x18      ; 24 字节：arr[16] + s + i
  ;—— 数组初始化：逐元素写入栈 ——
  mov   DWORD PTR [ebp-0x18], 0xa   ; arr[0] = 10
  mov   DWORD PTR [ebp-0x14], 0x14   ; arr[1] = 20
  mov   DWORD PTR [ebp-0x10], 0x1e   ; arr[2] = 30
  mov   DWORD PTR [ebp-0xc], 0x28    ; arr[3] = 40
  mov   DWORD PTR [ebp-0x4], 0x0     ; s = 0
  mov   DWORD PTR [ebp-0x8], 0x0     ; i = 0
  jmp   401040
00401032 ;—— 循环体 ——
  mov   eax, [ebp-0x8]             ; eax = i
  mov   eax, [ebp+eax*4-0x18]      ; eax = arr[i] ★关键寻址
  add   [ebp-0x4], eax              ; s += arr[i]
  add   [ebp-0x8], 0x1             ; i++
00401040 ;—— 循环条件 ——
  cmp   [ebp-0x8], 0x3
  jle   401032                   ; i <= 3 则继续
  mov   edx, [ebp-0x4]             ; s
  mov   eax, [ebp+0x8]             ; n
  add   eax, edx
  leave
  ret

arr[i] 的机器码地址 = ebp − 0x18 + i × 4
mov eax, [ebp + eax*4 − 0x18]
  └─基址─┘   └─变址─┘ └─偏移─┘
两个细节：① 编译器把 i<4 优化成 cmp i,3 / jle，语义等价；② 数组下标不"乘 4"，而是直接进 eax*4 比例变址——硬件一次算完，这是识别数组访问的黄金标志。
逆向识别点：见到 [ebp+reg*4-0xN] 或 [reg+reg*4] 这类"寄存器×4 再加减常量"→ 几乎可以断定在访问 int 数组（char 数组是 ×1，short 是 ×2）。配合循环计数器和 cmp/jle 结构，还原一个数组遍历非常快。
03 D4-3 指针传参：取址、解引用、指针算术 d3_ptr.c → d3_ptr.exe @ 0x401000

程序（节选）

void swap(int *a, int *b) {
    int t = *a;    // 解引用读
    *a = *b;       // 解引用写
    *b = t;
}
int main(void) {
    int x = 3, y = 5;
    int *p = &x;   // 指针变量存 x 的地址
    swap(&x, &y);  // 传地址
    p = p + 1;    // 指针算术
    return x - y;
}
32 位反汇编（节选）
00401000 <_swap>:
  push  ebp
  mov   ebp, esp
  sub   esp, 0x4            ; t 占 4 字节
  ;—— t = *a：先取地址，再按地址取值 ——
  mov   eax, [ebp+0x8]      ; eax = a（a 里存的是地址）
  mov   eax, [eax]            ; eax = *a（解引用）★两跳访存
  mov   [ebp-0x4], eax       ; t = eax
  ;—— *a = *b ——
  mov   eax, [ebp+0xc]        ; eax = b
  mov   edx, [eax]              ; edx = *b
  mov   eax, [ebp+0x8]
  mov   [eax], edx              ; *a = edx
  ;—— *b = t ——
  mov   eax, [ebp+0xc]
  mov   edx, [ebp-0x4]
  mov   [eax], edx              ; *b = t
  leave
  ret

00401023 <_main>:
  sub   esp, 0x14            ; x/y/p + 参数区
  mov   [ebp-0x8], 0x3       ; x = 3
  mov   [ebp-0xc], 0x5       ; y = 5
  lea   eax, [ebp-0x8]       ; eax = &x ★取地址
  mov   [ebp-0x4], eax       ; p = &x
  lea   eax, [ebp-0xc]       ; eax = &y
  mov   [esp+0x4], eax       ; 参数2 = &y
  lea   eax, [ebp-0x8]       ; eax = &x
  mov   [esp], eax             ; 参数1 = &x
  call  _swap
  add   [ebp-0x4], 0x4        ; p = p+1 → 实际 +4 字节！
  mov   edx, [ebp-0x8]        ; x（swap 后 = 5）
  mov   eax, [ebp-0xc]        ; y（swap 后 = 3）
  sub   edx, eax               ; x - y = 2
  leave
  ret

lea vs mov：lea eax,[ebp-8] 把"地址值"放进 eax（不访存）；mov eax,[eax] 才按地址取值。见到 lea reg,[ebp-0xN] 后紧跟 mov [esp],reg → 就是在传地址，即 C 里的 &var。
指针 +1 不是地址 +1：add [ebp-4], 0x4 —— 因为 p 是 int*，一个 int 占 4 字节，所以 p+1 在机器码里就是地址 +4。char* 的 +1 才是 +1。逆向还原时，看到指针加常量，要除以元素大小反推类型。
逆向识别点：swap 里 mov eax,[ebp+8] → mov eax,[eax] 的"取地址→再访存"两跳，就是 C 的 *p。值传递的函数绝不会出现这种双层访存。

07 自测：看到汇编能还原吗 先自己想，再点开答案
Q1 一段代码里出现 mov eax, ds:0x403010 且全程没有对 ebp 的访问。它大概率是什么？
答案
全局（或静态）变量。固定绝对地址 + 无栈帧 = 镜像数据段里的对象，C 里对应文件作用域变量。
Q2 看到 mov eax,[ebp+0x8] → mov eax,[eax] 连续两条指令，还原成 C 是什么？
答案
解引用：int t = *a;。第一条从栈取指针值（地址），第二条按地址取值。值传递的代码不会出现这种双层访存。
Q3 循环体里出现 mov eax,[ebp+ecx*4-0x20]，ecx 从 0 递增。这是什么数据结构？
答案
int 数组遍历。×4 是元素大小（sizeof(int)），-0x20 是数组首元素相对 ebp 的偏移，ecx 是下标 i。等价于 arr[i]。
Q4 函数 prologue 是 sub esp, 0x18，函数里访问了 [ebp-0x18] 到 [ebp-0xc] 四个 4 字节槽位。它可能是？
答案
一个 int arr[4]（16 字节连续槽位），也可能是 4 个相邻的局部 int。区分方法：看访问时是"固定偏移逐个访问"（局部变量）还是"带变址 reg*4"（数组下标）。
Q5 add DWORD PTR [ebp-0x4], 0x4，对应 C 里的 p = p + 1。p 是什么类型？
答案
int *。指针 +1 在机器码里加的是元素大小：+4 = int*，+1 = char*，+2 = short*，+8 = double* 或 long long*。反向可据此推断指针指向类型。
